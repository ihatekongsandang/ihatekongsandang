import * as React from "react"
;(globalThis as unknown as { React: typeof React }).React = React
// 클래식 JSX 변환(tsx 기본)에서 컴포넌트 파일이 React 전역을 찾으므로 렌더 전에 넣어 둔다.
import { renderToStaticMarkup } from 'react-dom/server'
import { TagNav } from '@/components/tag-nav'
const mk = (n: number) => Array.from({ length: n }, (_, i) => ({ tag: `t${String(i + 1).padStart(2, '0')}`, count: 100 - i }))
const sum = (html: string) => {
  const hrefs = [...html.matchAll(/href="([^"]+)"/g)].map((m) => m[1])
  const cur = [...html.matchAll(/<a[^>]*aria-current="page"[^>]*href="([^"]+)"/g)].map((m) => m[1])
  const all = /전체 태그 보기 \((?:<!-- -->)?(\d+)/.exec(html)?.[1]
  return { links: hrefs.length, chips: hrefs.filter((h) => h.startsWith('/tag/')).length, first: hrefs[0], last: hrefs.at(-1), current: cur, allCount: all }
}
const cases: [string, ReturnType<typeof mk>, string | undefined][] = [
  ['0 tags', mk(0), undefined],
  ['5 tags home', mk(5), undefined],
  ['5 tags active t03', mk(5), 't03'],
  ['12 tags home', mk(12), undefined],
  ['20 tags active=12th', mk(20), 't12'],
  ['20 tags active=13th', mk(20), 't13'],
  ['20 tags active=last', mk(20), 't20'],
  ['20 tags active=unknown', mk(20), 'zz'],
  ['encoding', [{ tag: '6-25전쟁', count: 3 }, { tag: 'a b/c?d#e%', count: 1 }], 'a b/c?d#e%'],
]
const input = mk(20); const snapshot = JSON.stringify(input)
renderToStaticMarkup(<TagNav tags={input} activeTag="t15" />)
for (const [name, tags, active] of cases) {
  const html = renderToStaticMarkup(<TagNav tags={tags} activeTag={active} />)
  console.log(name.padEnd(24), html === '' ? 'null(렌더 없음)' : JSON.stringify(sum(html)))
}
console.log('input mutated:', JSON.stringify(input) !== snapshot)
